### Migrate responses.RequestsMock unittest setUp to respx.mock()

Source: https://github.com/lundberg/respx/blob/master/docs/migrate.md

Shows how to replace responses.RequestsMock with respx.mock() in a unittest setUp method. Both use start() and stop() with addCleanup to ensure cleanup.

```python
def setUp(self):
    self.responses = responses.RequestsMock()
    self.responses.start()
    self.addCleanup(self.responses.stop)

```

```python
def setUp(self):
    self.respx_mock = respx.mock()
    self.respx_mock.start()
    self.addCleanup(self.respx_mock.stop)

```

### Example — lookups

Source: https://github.com/lundberg/respx/blob/master/_autodocs/api-reference/patterns.md

Use respx.get with url__regex, params, and headers__contains to mock a GET request. The example asserts a 200 response with JSON body.

```python
import httpx
import respx

respx.get(
    url__regex=r"https://example\.org/.*",
    params={"page": "2"},
    headers__contains={"X-Api-Key": "secret"},
).respond(200, json={"items": []})

response = httpx.get(
    "https://example.org/list",
    params={"page": "2"},
    headers={"X-Api-Key": "secret"},
)
assert response.status_code == 200
```

### Basic usage with global router

Source: https://github.com/lundberg/respx/blob/master/_autodocs/api-reference/module-api.md

Starts the respx global router, mocks a GET request to example.org to return a 204 response, performs the request with httpx, asserts the response status and that a call was made, then stops and clears the router.

```python
import httpx
import respx

respx.start()
respx.get("https://example.org/").mock(return_value=httpx.Response(204))
response = httpx.get("https://example.org/")
assert response.status_code == 204
assert respx.calls.called
respx.stop(clear=True, reset=True)
```

### unittest: Reuse setUpClass and setUp for mocked API

Source: https://github.com/lundberg/respx/blob/master/docs/examples.md

Defines a mixin class that sets up a mocked API in `setUpClass` and starts/stops it in `setUp`/`addCleanup`, allowing reuse across test cases. Requires `respx` and `httpx`.

```python
# testcases.py
import respx

from httpx import Response


class MockedAPIMixin:
    @classmethod
    def setUpClass(cls):
        cls.mocked_api = respx.mock(
            base_url="https://foo.bar", assert_all_called=False
        )
        users_route = cls.mocked_api.get("/users/", name="list_users")
        users_route.return_value = Response(200, json=[])
        ...

    def setUp(self):
        self.mocked_api.start()
        self.addCleanup(self.mocked_api.stop)
```

### Base URL Configuration with respx.mock(base_url=...)

Source: https://github.com/lundberg/respx/blob/master/docs/guide.md

Configures the router with a base_url so routes can be added with relative paths. The example uses an async client and mocks a response.

```python
import httpx
import respx

from httpx import Response


@respx.mock(base_url="https://example.org/api/")
async def test_something(respx_mock):
    async with httpx.AsyncClient(base_url="https://example.org/api/") as client:
        respx_mock.get("/baz/").mock(return_value=Response(200, text="Baz"))
        response = await client.get("/baz/")
        assert response.text == "Baz"

```

### Marker-based configuration examples

Source: https://github.com/lundberg/respx/blob/master/_autodocs/api-reference/pytest-plugin.md

Two examples showing how to use the @pytest.mark.respx marker to configure a fresh MockRouter per test. The first sets base_url, the second enables strict assertion options; leaving a route uncalled in the second raises on fixture teardown.

```python
import httpx
import pytest


@pytest.mark.respx(base_url="https://foo.bar")
def test_with_marker(respx_mock):
    respx_mock.get("/baz/").mock(return_value=httpx.Response(204))
    response = httpx.get("https://foo.bar/baz/")
    assert response.status_code == 204
```

```python
@pytest.mark.respx(assert_all_called=True, using="httpx")
def test_strict(respx_mock):
    respx_mock.get("/must-be-called/").respond(200)
    # leaving this route uncalled raises on fixture teardown
```

### Example — direct pattern objects and errors

Source: https://github.com/lundberg/respx/blob/master/_autodocs/api-reference/patterns.md

Constructs a Route with Method, Host, and Path patterns, adds it to a Router with assert_all_mocked=True, and tests a mismatched method raises AssertionError. Also shows that M(badkey="x") raises KeyError for invalid pattern keys.

```python
import re
import httpx
from respx import Router
from respx.models import Route
from respx.patterns import Host, M, Method, Path

route = Route(
    Method("get"),
    Host(re.compile(r"api\..+\.com"), lookup=Lookup.REGEX),
    Path("/v1/", lookup=Lookup.STARTS_WITH),
)
router = Router(assert_all_mocked=True)
router.add(route)

try:
    router.handler(httpx.Request("POST", "https://api.foo.com/v1/items"))
except AssertionError:  # AllMockedAssertionError: method did not match
    pass

try:
    M(badkey="x")  # 'badkey' is not a registered pattern key
except KeyError as error:
    assert "is not a valid Pattern" in str(error)
```

### Sync App Example: Mock with WSGIHandler

Source: https://github.com/lundberg/respx/blob/master/docs/guide.md

Mocks a route with a Flask app using WSGIHandler as side effect. Requires Flask and httpx. The test asserts the JSON response and that the route was called.

```python
import httpx
import respx

from flask import Flask

app = Flask("foobar")


@app.route("/baz/")
def baz():
    return {"ham": "spam"}


@respx.mock(base_url="https://foo.bar/")
def test_baz(respx_mock):
    app_route = respx_mock.route().mock(side_effect=WSGIHandler(app))
    response = httpx.get("https://foo.bar/baz/")
    assert response.json() == {"ham": "spam"}
    assert app_route.called
```

### Async App Example: Mock with ASGIHandler

### WSGI app as side effect (sync)

import httpx
import respx
from respx import WSGIHandler

def wsgi_app(environ, start_response):
    start_response("200 OK", [("Content-Type", "text/plain")])

response = httpx.get("https://example.org/app")

### URL pattern: shorthand and eq, regex, startswith lookups

### Mocking with %, respond(), and side-effect iteration

Source: https://github.com/lundberg/respx/blob/master/_autodocs/api-reference/Route.md

Shows how to use the % operator to set a status code, a dict of response kwargs, or a raw httpx.Response. Also demonstrates stacking side effects with a list (404, then 500, then 204) and a callable side effect that raises an httpx.ConnectError.

```python
import httpx
import respx

route = respx.get("https://example.org/")
route % 204                                            # int -> status code
route % {"status_code": 200, "json": {"ok": True}}     # dict -> Response(**dict)
route % httpx.Response(500)                            # raw Response

# Stacked side effects (iterable): 404, then 500, then 204
respx.get("https://example.org/api/").mock(
    side_effect=[httpx.Response(404), httpx.Response(500), httpx.Response(204)]
)
assert httpx.get("https://example.org/api/").status_code == 404

# Callable side effect raising an httpx error
def boom(request):
    raise httpx.ConnectError("down")

try:
    httpx.get("https://example.org/api/")
except httpx.ConnectError as error:
    assert str(error) == "down"
```

### TryTransport

Source: https://github.com/lundberg/respx/blob/master/_autodocs/api-reference/transports.md

Documentation for the TryTransport class, a dual sync/async transport that tries each of its transports in order, swallowing PassThrough exceptions and falling through to the next transport.

```APIDOC
## TryTransport(BaseTransport, AsyncBaseTransport)

### Description
A dual sync/async transport that tries each of its transports in order, swallowing `PassThrough` (raised by the router when a route is marked for pass-through) and falling through to the next transport. This is how `HTTPXMocker` chains the mock transport in front of the real transport.

### Constructor
`__init__(self, transports: List[Union[BaseTransport, AsyncBaseTransport]]) -> None`

### Parameters
- **transports** (List[Union[BaseTransport, AsyncBaseTransport]]) - Required - Transports to try, in order

### Methods

#### handle_request(self, request: httpx.Request) -> httpx.Response
Synchronously tries each transport in order, returning the first response that does not raise `PassThrough`. If all transports raise `PassThrough`, raises `RuntimeError`.

#### handle_async_request(self, request: httpx.Request) -> httpx.Response
Asynchronously tries each transport in order, returning the first response that does not raise `PassThrough`. If all transports raise `PassThrough`, raises `RuntimeError`.

### Raises
- `RuntimeError()` — when no transport handles the request (only reachable if every transport raises `PassThrough`).
```

### Mock without Patching HTTPX using MockTransport

Source: https://github.com/lundberg/respx/blob/master/docs/guide.md

Use httpx.MockTransport with a RESPX router as handler to mock HTTPX without patching. The router's handler and async_handler are used for sync and async clients respectively. Note that you must manually call <router>.assert_all_called() to assert all routes were called.

```python
import httpx
import respx


router = respx.Router()
router.post("https://example.org/") % 404


def test_client():
    mock_transport = httpx.MockTransport(router.handler)
    with httpx.Client(transport=mock_transport) as client:
        response = client.post("https://example.org/")
        assert response.status_code == 404


def test_client():
    mock_transport = httpx.MockTransport(router.async_handler)

### TryTransport class implementation

Source: https://github.com/lundberg/respx/blob/master/_autodocs/api-reference/transports.md

Defines TryTransport, a dual sync/async transport that tries each transport in order, catching PassThrough to fall through. Raises RuntimeError if no transport handles the request.

```python
class TryTransport(BaseTransport, AsyncBaseTransport):
    def __init__(
        self, transports: List[Union[BaseTransport, AsyncBaseTransport]]
    ) -> None:
        self.transports = transports

    def handle_request(self, request: httpx.Request) -> httpx.Response:
        for transport in self.transports:
            try:
                transport = cast(BaseTransport, transport)
                return transport.handle_request(request)
            except PassThrough:
                continue

        raise RuntimeError()  # pragma: nocover

    async def handle_async_request(self, request: httpx.Request) -> httpx.Response:
        for transport in self.transports:
            try:
                transport = cast(AsyncBaseTransport, transport)
                return await transport.handle_async_request(request)
            except PassThrough:
                continue

        raise RuntimeError()  # pragma: nocover
```

### Route.resolve(request, **kwargs)

Source: https://github.com/lundberg/respx/blob/master/_autodocs/api-reference/Route.md

Resolves a request to a response, using side effect, return value, or auto-mocked response.

```APIDOC
## `resolve(request, **kwargs)`

### Description
Resolves a request to a response. Resolution order: `self._side_effect` first (`_resolve_side_effect`), else `self._return_value`, else an auto-mocked `httpx.Response(200, request=request)`. Responses that have no `_request` set are cloned per-request for immutability via `clone_response`.

### Signature

### Parameters

- **kwargs** (Any) - Optional - Additional context (e.g., matched regex groups).

### Returns
`RouteResultTypes` - The resolved response or request.

### Side-effect resolution semantics

| Side effect value | Behavior |
|---|---|
| `Exception` instance | `SideEffectError(self, origin=effect)` is raised; the router re-raises `origin` (see [errors.md](../errors.md)) |
| `Exception` class | `SideEffectError(self, origin=effect("Mock Error", request=request))` when it subclasses `httpx.RequestError`, else `effect()`; result re-raised as `origin` |
| callable | Called as `effect(request, **matched_context)`; a `route` kwarg is injected when the callable accepts it (a `UserWarning` is emitted for using the reserved word `route`); async results are awaited by `Router.aresolve`. Returns `None` for a non-match, or a `httpx.Response`/`httpx.Request` (raise `TypeError` otherwise) |
| `httpx.Response` / anything else | Returned as the resolved response |

```python
class RouteList:
    _routes: List[Route]
    _names: Dict[str, Route]

    def __init__(self, routes: Optional["RouteList"] = None) -> None:
        if routes is None:
            self._routes = []
            self._names = {}
        else:
            self._routes = list(routes._routes)
            self._names = dict(routes._names)

    def __repr__(self) -> str:
        return repr(self._routes)  # pragma: nocover

    def __iter__(self) -> Iterator[Route]:
        return iter(self._routes)

    def __bool__(self) -> bool:
        return bool(self._routes)

    def __len__(self) -> int:
        return len(self._routes)

    def __contains__(self, name: str) -> bool:
        return name in self._names

    def __getitem__(self, key: Union[int, str]) -> Route:
        if isinstance(key, int):
            return self._routes[key]
        else:
            return self._names[key]

    def __setitem__(self, i: slice, routes: "RouteList") -> None:
        """
        Re-set all routes to given routes.
        """
        if (i.start, i.stop, i.step) != (None, None, None):
            raise TypeError("Can't slice assign routes")
        self._routes = list(routes._routes)
        self._names = dict(routes._names)

    def clear(self) -> None:
        self._routes.clear()
        self._names.clear()

    def add(self, route: Route, name: Optional[str] = None) -> Route:
        ...

    def pop(self, name, default=...):
        ...

### SetCookie NamedTuple with __new__ for cookie serialization

Source: https://github.com/lundberg/respx/blob/master/_autodocs/api-reference/utils.md

Defines the SetCookie NamedTuple used to build a serialized Set-Cookie header value. Accepts optional attributes; datetime expires values are formatted as RFC 1123, and same_site="None" forces secure=True. Bare boolean attributes are rendered without =... .

```python
Self = TypeVar("Self", bound="SetCookie")


class SetCookie(
    NamedTuple(
        "SetCookie",
        [
            ("header_name", Literal["Set-Cookie"]),
            ("header_value", str),
        ],
    )
):
    def __new__(
        cls: Type[Self],
        name: str,
        value: str,
        *,
        path: Optional[str] = None,
        domain: Optional[str] = None,
        expires: Optional[Union[str, datetime]] = None,
        max_age: Optional[int] = None,
        http_only: bool = False,
        same_site: Optional[Literal["Strict", "Lax", "None"]] = None,
        secure: bool = False,
        partitioned: bool = False,
    ) -> Self:
        """
        https://developer.mozilla.org/en-US/docs/Web/HTTP/Headers/Set-Cookie#syntax
        """
        attrs: Dict[str, Union[str, bool]] = {name: value}
        if path is not None:
            attrs["Path"] = path
        if domain is not None:
            attrs["Domain"] = domain
        if expires is not None:
            if isinstance(expires, datetime):  # pragma: no branch
                expires = expires.strftime("%a, %d %b %Y %H:%M:%S GMT")
            attrs["Expires"] = expires
        if max_age is not None:
            attrs["Max-Age"] = str(max_age)
        if http_only:
            attrs["HttpOnly"] = True
        if same_site is not None:
            attrs["SameSite"] = same_site
            if same_site == "None":  # pragma: no branch
                secure = True
        if secure:
            attrs["Secure"] = True
        if partitioned:
            attrs["Partitioned"] = True

        string = "; ".join(
            _name if _value is True else f"{_name}={_value}"
            for _name, _value in attrs.items()
        )
        self = super().__new__(cls, "Set-Cookie", string)
        return self
```

### MultiItems defaultdict class

Source: https://github.com/lundberg/respx/blob/master/_autodocs/api-reference/utils.md

A defaultdict whose default factory is tuple, holding multiple values per key. Used to represent form data / files in the Data and Files patterns. Provides get_list, multi_items, and append methods.

```python
class MultiItems(defaultdict):
    def __init__(self, values: Optional[Iterable[Tuple[str, Any]]] = None) -> None:
        super().__init__(tuple)
        if values is not None:
            for key, value in values:
                if isinstance(value, (tuple, list)):
                    self[key] += tuple(value)  # Convert list to tuple and extend
                else:
                    self[key] += (value,)  # Extend with value

    def get_list(self, key: str) -> List[Any]:
        return list(self[key])

    def multi_items(self) -> List[Tuple[str, str]]:
        return [(key, value) for key, values in self.items() for value in values]

    def append(self, key: str, value: Any) -> None:
        self[key] += (value,)
```
